MariaDB(MySQL) 환경에서 약관의 계층 구조와 버전 관리를 완벽하게 지원할 수 있는 테이블 설계 쿼리입니다. 각 테이블과 컬럼에 상세한 COMMENT를 추가하였습니다.

1. 약관 마스터 테이블 (Terms)
어떤 종류의 약관(서비스 이용약관, 개인정보 처리방침 등)인지 정의합니다.

SQL
CREATE TABLE `terms` (
    `terms_id` INT UNSIGNED AUTO_INCREMENT PRIMARY KEY COMMENT '약관 고유 식별자',
    `title` VARCHAR(255) NOT NULL COMMENT '약관 명칭 (예: 서비스 이용약관)',
    `code` VARCHAR(50) UNIQUE NOT NULL COMMENT '약관 식별 코드 (예: SERVICE_TERMS, PRIVACY_POLICY)',
    `is_required` TINYINT(1) DEFAULT 1 COMMENT '필수 동의 여부 (1: 필수, 0: 선택)',
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '최초 생성 일시',
    `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '수정 일시'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='약관 종류 마스터 테이블';
2. 약관 섹션(조항) 테이블 (Term_Sections)
약관을 '조(Article)' 또는 '항(Paragraph)' 단위로 나누어 저장하는 테이블입니다.

SQL
CREATE TABLE `term_sections` (
    `section_id` INT UNSIGNED AUTO_INCREMENT PRIMARY KEY COMMENT '섹션 고유 식별자',
    `terms_id` INT UNSIGNED NOT NULL COMMENT '소속 약관 ID (FK)',
    `parent_id` INT UNSIGNED DEFAULT NULL COMMENT '상위 섹션 ID (조-항 관계를 위한 셀프 참조)',
    `version` VARCHAR(20) NOT NULL COMMENT '해당 섹션의 버전 (예: 1.0, 1.1)',
    `title` VARCHAR(255) COMMENT '조항 제목 (예: 제1조 목적)',
    `content` TEXT NOT NULL COMMENT '조항 본문 내용 (최소 단위)',
    `order_no` INT UNSIGNED NOT NULL DEFAULT 1 COMMENT '노출 순서',
    `status` ENUM('ACTIVE', 'INACTIVE', 'DELETED') DEFAULT 'ACTIVE' COMMENT '상태 (활성, 비활성, 삭제)',
    `is_latest` TINYINT(1) DEFAULT 1 COMMENT '현재 최신 버전 여부 (1: 최신, 0: 과거버전)',
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '생성 일시',
    `created_by` VARCHAR(50) COMMENT '작성자/수정자 ID',
    CONSTRAINT `fk_sections_terms` FOREIGN KEY (`terms_id`) REFERENCES `terms` (`terms_id`) ON DELETE CASCADE,
    CONSTRAINT `fk_sections_parent` FOREIGN KEY (`parent_id`) REFERENCES `term_sections` (`section_id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='약관 구성 요소 및 실시간 본문 테이블';
3. 변경 이력 관리 테이블 (Term_History)
누가, 언제, 왜 변경했는지에 대한 상세 로그를 기록합니다.

SQL
CREATE TABLE `term_history` (
    `history_id` INT UNSIGNED AUTO_INCREMENT PRIMARY KEY COMMENT '이력 고유 식별자',
    `section_id` INT UNSIGNED NOT NULL COMMENT '변경된 섹션 ID (FK)',
    `change_type` ENUM('INSERT', 'UPDATE', 'DELETE') NOT NULL COMMENT '변경 유형',
    `prev_content` TEXT COMMENT '수정 전 본문 데이터',
    `new_content` TEXT COMMENT '수정 후 본문 데이터',
    `version_note` VARCHAR(500) COMMENT '수정 사유 및 버전 변경 사항 비고',
    `changed_at` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '변경 처리 일시',
    `operator_id` VARCHAR(50) COMMENT '처리 담당자 식별자',
    CONSTRAINT `fk_history_sections` FOREIGN KEY (`section_id`) REFERENCES `term_sections` (`section_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='약관 수정/삭제 이력 로그 테이블';
💡 운영을 위한 설계 포인트
셀프 참조(parent_id): * parent_id가 NULL이면 **'조'**가 되고, 특정 조의 section_id를 참조하면 그 아래의 **'항'**이 됩니다. 이를 통해 N단계 계층 구조를 만들 수 있습니다.

최신 버전 관리(is_latest): * 조회 성능을 위해 현재 서비스에 노출되어야 할 행에는 is_latest = 1을 부여합니다. 새로운 버전이 등록되면 기존 행은 0으로 업데이트하고 새 행을 추가합니다.

무결성 유지: * Term_History 테이블을 통해 단순 수정뿐만 아니라 '왜' 바뀌었는지에 대한 법적 근거(버전 노트)를 남기도록 설계했습니다.
