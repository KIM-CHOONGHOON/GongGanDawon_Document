5. 상품 마스터 테이블 (funeral_products)
sqlCREATE TABLE funeral_products (
    product_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '상품 ID',
    product_code VARCHAR(20) UNIQUE NOT NULL COMMENT '상품 코드',
    major_category_code VARCHAR(2) NOT NULL COMMENT '대분류 코드',
    middle_category_code VARCHAR(4) NOT NULL COMMENT '중분류 코드',
    minor_category_code VARCHAR(6) NOT NULL COMMENT '소분류 코드',
    product_name VARCHAR(200) NOT NULL COMMENT '상품명',
    specification TEXT COMMENT '규격/사양',
    description TEXT COMMENT '서비스 내용/설명',
    material VARCHAR(100) COMMENT '재질',
    size VARCHAR(100) COMMENT '크기',
    color VARCHAR(50) COMMENT '색상',
    manufacturer VARCHAR(100) COMMENT '제조사/공급사',
    country_of_origin VARCHAR(50) COMMENT '원산지',
    
    -- 기본 단위 (대표 단위)
    default_unit_code VARCHAR(10) NOT NULL COMMENT '기본 단위 코드',
    
    -- 옵션
    has_multiple_units BOOLEAN DEFAULT FALSE COMMENT '복수 단위 여부',
    is_required BOOLEAN DEFAULT FALSE COMMENT '필수 품목 여부',
    is_optional BOOLEAN DEFAULT TRUE COMMENT '선택 품목 여부',
    is_package BOOLEAN DEFAULT FALSE COMMENT '패키지 여부',
    
    display_order INT COMMENT '표시 순서',
    image_url VARCHAR(255) COMMENT '이미지 URL',
    is_active BOOLEAN DEFAULT TRUE COMMENT '사용 여부',
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (major_category_code) REFERENCES funeral_major_categories(major_category_code),
    FOREIGN KEY (middle_category_code) REFERENCES funeral_middle_categories(middle_category_code),
    FOREIGN KEY (minor_category_code) REFERENCES funeral_minor_categories(minor_category_code),
    FOREIGN KEY (default_unit_code) REFERENCES funeral_unit_codes(unit_code)
) COMMENT='장례용품 상품 마스터';

-- 인덱스
CREATE INDEX idx_major_category ON funeral_products(major_category_code);
CREATE INDEX idx_middle_category ON funeral_products(middle_category_code);
CREATE INDEX idx_minor_category ON funeral_products(minor_category_code);
CREATE INDEX idx_active ON funeral_products(is_active);
6. 상품별 단위 가격 테이블 (funeral_product_unit_prices)
sqlCREATE TABLE funeral_product_unit_prices (
    id INT PRIMARY KEY AUTO_INCREMENT COMMENT 'ID',
    product_id INT NOT NULL COMMENT '상품 ID',
    unit_code VARCHAR(10) NOT NULL COMMENT '단위 코드',
    unit_value DECIMAL(10,2) COMMENT '단위 값 (수량, 거리 등)',
    standard_price DECIMAL(12,2) NOT NULL COMMENT '표준 단가',
    
    -- 추가 정보
    min_quantity DECIMAL(10,2) DEFAULT 1 COMMENT '최소 수량/값',
    max_quantity DECIMAL(10,2) COMMENT '최대 수량/값',
    price_note TEXT COMMENT '가격 특이사항',
    
    display_order INT COMMENT '표시 순서',
    is_active BOOLEAN DEFAULT TRUE COMMENT '사용 여부',
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (product_id) REFERENCES funeral_products(product_id),
    FOREIGN KEY (unit_code) REFERENCES funeral_unit_codes(unit_code),
    UNIQUE KEY uk_product_unit (product_id, unit_code, unit_value)
) COMMENT='상품별 단위 가격';

-- 인덱스
CREATE INDEX idx_product ON funeral_product_unit_prices(product_id);
CREATE INDEX idx_unit ON funeral_product_unit_prices(unit_code);
7. 장례식장별 상품 단위 가격 테이블 (funeral_home_product_prices)
sqlCREATE TABLE funeral_home_product_prices (
    id INT PRIMARY KEY AUTO_INCREMENT COMMENT 'ID',
    funeral_home_id INT NOT NULL COMMENT '장례식장 ID',
    product_id INT NOT NULL COMMENT '상품 ID',
    unit_code VARCHAR(10) NOT NULL COMMENT '단위 코드',
    unit_value DECIMAL(10,2) COMMENT '단위 값',
    
    -- 가격 정보
    sale_price DECIMAL(12,2) NOT NULL COMMENT '판매 단가',
    discount_rate DECIMAL(5,2) COMMENT '할인율',
    
    -- 수량/범위 제한
    min_quantity DECIMAL(10,2) DEFAULT 1 COMMENT '최소 수량',
    max_quantity DECIMAL(10,2) COMMENT '최대 수량',
    
    -- 추가 정보
    is_available BOOLEAN DEFAULT TRUE COMMENT '제공 여부',
    is_included_in_package BOOLEAN DEFAULT FALSE COMMENT '패키지 포함 여부',
    note TEXT COMMENT '특이사항',
    
    display_order INT COMMENT '표시 순서',
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (funeral_home_id) REFERENCES funeral_homes(id),
    FOREIGN KEY (product_id) REFERENCES funeral_products(product_id),
    FOREIGN KEY (unit_code) REFERENCES funeral_unit_codes(unit_code),
    UNIQUE KEY uk_home_product_unit (funeral_home_id, product_id, unit_code, unit_value)
) COMMENT='장례식장별 상품 단위 가격';

-- 인덱스
CREATE INDEX idx_funeral_home ON funeral_home_product_prices(funeral_home_id);
CREATE INDEX idx_product ON funeral_home_product_prices(product_id);
CREATE INDEX idx_available ON funeral_home_product_prices(is_available);

조회 뷰 (View)
sql-- 전체 상품 정보 통합 뷰
CREATE VIEW v_funeral_products_full AS
SELECT 
    p.product_id,
    p.product_code,
    p.product_name,
    
    -- 카테고리 정보
    maj.major_category_code,
    maj.major_category_name,
    mid.middle_category_code,
    mid.middle_category_name,
    min.minor_category_code,
    min.minor_category_name,
    
    -- 상품 정보
    p.specification,
    p.description,
    p.material,
    p.size,
    p.color,
    
    -- 기본 단위
    p.default_unit_code,
    uc.unit_name AS default_unit_name,
    p.has_multiple_units,
    
    -- 기타
    p.is_required,
    p.is_optional,
    p.is_active
FROM funeral_products p
INNER JOIN funeral_major_categories maj ON p.major_category_code = maj.major_category_code
INNER JOIN funeral_middle_categories mid ON p.middle_category_code = mid.middle_category_code
INNER JOIN funeral_minor_categories min ON p.minor_category_code = min.minor_category_code
INNER JOIN funeral_unit_codes uc ON p.default_unit_code = uc.unit_code
WHERE p.is_active = TRUE;

-- 상품별 단위 가격 뷰
CREATE VIEW v_product_unit_prices AS
SELECT 
    p.product_id,
    p.product_code,
    p.product_name,
    maj.major_category_name,
    mid.middle_category_name,
    min.minor_category_name,
    
    -- 단위별 가격
    pup.id AS price_id,
    pup.unit_code,
    uc.unit_name,
    uc.unit_type,
    pup.unit_value,
    pup.standard_price,
    pup.min_quantity,
    pup.max_quantity,
    pup.price_note
FROM funeral_product_unit_prices pup
INNER JOIN funeral_products p ON pup.product_id = p.product_id
INNER JOIN funeral_major_categories maj ON p.major_category_code = maj.major_category_code
INNER JOIN funeral_middle_categories mid ON p.middle_category_code = mid.middle_category_code
INNER JOIN funeral_minor_categories min ON p.minor_category_code = min.minor_category_code
INNER JOIN funeral_unit_codes uc ON pup.unit_code = uc.unit_code
WHERE pup.is_active = TRUE
ORDER BY p.product_code, uc.display_order, pup.unit_value;

-- 장례식장별 상품 가격 뷰
CREATE VIEW v_funeral_home_product_prices AS
SELECT 
    fh.id AS funeral_home_id,
    fh.name AS funeral_home_name,
    
    p.product_id,
    p.product_code,
    p.product_name,
    
    maj.major_category_name,
    mid.middle_category_name,
    min.minor_category_name,
    
    -- 단위 정보
    fhpp.unit_code,
    uc.unit_name,
    uc.unit_type,
    fhpp.unit_value,
    
    -- 가격 정보
    pup.standard_price,
    fhpp.sale_price,
    fhpp.discount_rate,
    ROUND((pup.standard_price - fhpp.sale_price) / pup.standard_price * 100, 2) AS actual_discount_rate,
    
    -- 수량 제한
    fhpp.min_quantity,
    fhpp.max_quantity,
    
    -- 기타
    fhpp.is_available,
    fhpp.is_included_in_package,
    fhpp.note
FROM funeral_home_product_prices fhpp
INNER JOIN funeral_homes fh ON fhpp.funeral_home_id = fh.id
INNER JOIN funeral_products p ON fhpp.product_id = p.product_id
INNER JOIN funeral_major_categories maj ON p.major_category_code = maj.major_category_code
INNER JOIN funeral_middle_categories mid ON p.middle_category_code = mid.middle_category_code
INNER JOIN funeral_minor_categories min ON p.minor_category_code = min.minor_category_code
INNER JOIN funeral_unit_codes uc ON fhpp.unit_code = uc.unit_code
LEFT JOIN funeral_product_unit_prices pup ON fhpp.product_id = pup.product_id 
    AND fhpp.unit_code = pup.unit_code 
    AND (fhpp.unit_value = pup.unit_value OR pup.unit_value IS NULL)
WHERE fhpp.is_available = TRUE
ORDER BY fh.name, maj.display_order, mid.display_order, min.display_order, p.product_code;

sql-- 1. 특정 장례식장의 수의 칭수별 가격 조회
SELECT 
    funeral_home_name,
    product_name,
    unit_name,
    unit_value,
    standard_price,
    sale_price,
    actual_discount_rate
FROM v_funeral_home_product_prices
WHERE funeral_home_id = 1
  AND middle_category_name = '수의'
ORDER BY unit_value;

-- 2. 운구차 거리별 가격 비교
SELECT 
    funeral_home_name,
    product_name,
    unit_name,
    unit_value,
    sale_price
FROM v_funeral_home_product_prices
WHERE minor_category_name LIKE '%운구차%'
ORDER BY funeral_home_name, unit_code, unit_value;

-- 3. 제단 형별 가격 조회
SELECT 
    product_name,
    unit_name,
    standard_price,
    price_note
FROM v_product_unit_prices
WHERE middle_category_name = '제단'
ORDER BY unit_value;

-- 4. 꽃 송이별 가격 조회
SELECT 
    product_name,
    CONCAT(unit_value, unit_name) AS quantity,
    standard_price,
    ROUND(standard_price / unit_value, 0) AS price_per_unit
FROM v_product_unit_prices
WHERE minor_category_name = '국화'
  AND unit_code = 'BUNCH'
ORDER BY unit_value;

-- 5. 상복 벌수별 가격 조회
SELECT 
    product_name,
    CONCAT(unit_value, unit_name) AS quantity,
    standard_price,
    ROUND(standard_price / unit_value, 0) AS price_per_piece
FROM v_product_unit_prices
WHERE middle_category_name = '상복'
  AND unit_code = 'PIECE'
ORDER BY unit_value;

-- 6. 버스 인승별 + 거리별 가격 조회
SELECT 
    funeral_home_name,
    product_name,
    unit_name,
    unit_value,
    sale_price,
    note
FROM v_funeral_home_product_prices
WHERE product_name LIKE '%버스%'
ORDER BY funeral_home_name, unit_type, unit_value;
